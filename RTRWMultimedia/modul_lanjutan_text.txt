namespace RTRWMultimedia.Database 
{ 
    public class Koneksi 
    { 
        public static SqlConnection GetConnection() 
        { 
            SqlConnection conn = 
            new SqlConnection( 
            @"Data Source=.\SQLEXPRESS; 
            Initial Catalog=DB_RTRW; 
            Integrated Security=True"); 
 
            return conn; 
        } 
    } 
}

private void btnLogin_Click(

{

    Koneksi.GetConnection();

    string sql = 
    "SELECT * FROM tb_user " + 
    "WHERE username=@user " + 
    "AND password=@pass"; 
 
    SqlCommand cmd = 
    new SqlCommand(sql, conn); 
 
    cmd.Parameters.AddWithValue(

    txtUsername.Text); 
 
    cmd.Parameters.AddWithValue( 
    "@pass", 
    txtPassword.Text); 
 
    SqlDataReader rd = 
    cmd.ExecuteReader();

        MessageBox.Show( 
        "Login Berhasil"); 
 
        frmDashboard f = 
        new frmDashboard();

        this.Hide();
